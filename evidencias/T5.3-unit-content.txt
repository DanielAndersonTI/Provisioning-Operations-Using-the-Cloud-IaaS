# /etc/systemd/system/app-cpu.service
[Unit]
Description=Atividade 3 - aplicacao cpu (porta 8000)
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=appsvc
Group=appsvc
WorkingDirectory=/opt/atividade3/apps
Environment=PYTHONUNBUFFERED=1
ExecStart=/usr/bin/python3 /opt/atividade3/apps/cpu/server.py --port 8000
Restart=on-failure
RestartSec=2
TimeoutStopSec=5
StandardOutput=append:/var/log/atividade3/cpu/app.log
StandardError=append:/var/log/atividade3/cpu/app.log

# Endurecimento. A aplicacao precisa apenas de rede, de CPU e do proprio /tmp
# privado; nada em /opt e reescrito em tempo de execucao.
NoNewPrivileges=yes
PrivateTmp=yes
ProtectHome=yes
ProtectSystem=full
ReadWritePaths=/var/log/atividade3/cpu

[Install]
WantedBy=multi-user.target

# /etc/systemd/system/app-cpu.service.d/volume-logs.conf
[Service]
StandardOutput=append:/mnt/atividade3-logs/cpu/app.log
StandardError=append:/mnt/atividade3-logs/cpu/app.log
ReadWritePaths=/var/log/atividade3/cpu /mnt/atividade3-logs/cpu
